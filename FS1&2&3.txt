#Q1.Write a JavaScript program to demonstrate all operations of the Set object using a student example. The program should store student roll numbers, add new student entries, display the total number of students, check whether a particular student roll number is present, remove a student roll number, traverse all student records, display the stored student values, and finally remove all student data from the Set.
const rolls = new Set([101, 102]);
rolls.add(103); // Add entry
console.log("Total:", rolls.size); // Total students
console.log("Exists 102?", rolls.has(102)); // Check presence
rolls.delete(101); // Remove entry
rolls.forEach(r => console.log("Roll:", r)); // Traverse/Display values
rolls.clear(); // Remove all
#Q2.Write a JavaScript program to demonstrate all operations of the Map object using an employee example. The program should store employee IDs as keys and employee details as values, add new employee records, display the total number of employees, check whether a particular employee ID exists, retrieve employee details using the employee ID, remove an employee record, traverse and display all employee records, display all stored keys and values, and finally remove all employee data from the Map.
const employees = new Map([[1, {name: "Alice"}]]);
employees.set(2, {name: "Bob"}); // Add record
console.log("Total:", employees.size); // Total count
console.log("Has ID 1?", employees.has(1)); // Check ID
console.log("Details ID 2:", employees.get(2)); // Retrieve
employees.delete(1); // Remove record
employees.forEach((val, key) => console.log(key, val)); // Traverse
employees.clear(); // Remove all

#Q4.Create a Shape class with a method area(), and extend it into a Rectangle class that overrides the area() method to calculate the area of a rectangle.
class Shape { area() { return 0; } }
class Rectangle extends Shape {
constructor(w, h) { super(); this.w = w; this.h = h; }
area() { return this.w * this.h; }
}
console.log(new Rectangle(5, 10).area()); // 50
#Q5.Write a program to demonstrate the use of the super() keyword.
class Parent { constructor(name) { this.name = name; } }
class Child extends Parent {
constructor(name, age) {
super(name); // Calls Parent constructor
this.age = age;
}
}
const kid = new Child("Tim", 10);
console.log(kid.name, kid.age);
